// ==========================================
// 1. BASE USERS & EXPLICIT ROLES
// ==========================================

Table users {
  user_id uuid [pk]
  email varchar [unique] // Username
  password_hash text
  full_name varchar
  phone_number varchar // Added as requested
  
  // 'owner', 'director', 'vice_director', 'educator', 'parent'
  role_type varchar 
  
  is_active boolean [default: true]
  created_at timestamp
}

// --- 1.1 NETWORK OWNER ---
Table owners {
  user_id uuid [pk, ref: - users.user_id] // 1-to-1 inheritance
  office_address text
}

// --- 1.2 CENTER DIRECTORS ---
Table directors {
  user_id uuid [pk, ref: - users.user_id]
  qualification_cert varchar
  hired_at date
}

// --- 1.3 VICE DIRECTORS (With Checkbox Permissions) ---
Table vice_directors {
  user_id uuid [pk, ref: - users.user_id]
  center_id uuid [ref: > centers.center_id] // The center they manage
  
  // JSON Checkboxes: { "can_edit_ratios": true, "can_view_finance": false }
  specific_permissions jsonb 
}

// --- 1.4 EDUCATORS ---
Table educators {
  user_id uuid [pk, ref: - users.user_id]
  center_id uuid [ref: > centers.center_id]
  
  color_code varchar // For calendar UI
  is_active_in_ratio boolean // Do they count towards legal limits?
}

// --- 1.5 PARENTS ---
Table parents {
  user_id uuid [pk, ref: - users.user_id]
  home_address text
  billing_reference varchar
}

// ==========================================
// 2. ORGANIZATION HIERARCHY
// ==========================================

Table organizations {
  org_id uuid [pk]
  name varchar
  owner_id uuid [ref: - owners.user_id] // Link to Owner Table
  created_at timestamp
}

Table centers {
  center_id uuid [pk]
  org_id uuid [ref: > organizations.org_id]
  name varchar
  address text
  timezone varchar
  
  // Link to the Main Director
  director_id uuid [ref: - directors.user_id] 
}

// Physical Rooms
Table rooms {
  room_id uuid [pk]
  center_id uuid [ref: > centers.center_id]
  name varchar // "Blue Room"
  fire_capacity int // Strict safety limit
}

// Logical Groups (The entity tracked for Ratios)
Table groups {
  group_id uuid [pk]
  room_id uuid [ref: > rooms.room_id] // Assigned physical room
  center_id uuid [ref: > centers.center_id]
  
  name varchar // "The Sunflowers"
  
  // NB Rules: 'UNDER_24M', 'MIXED_2_TO_4', 'SCHOOL_AGE_5_PLUS'
  ratio_category varchar 
  
  // --- REAL TIME STATS ---
  current_occupancy int
  current_educators int
  
  // The cumulative value (e.g. 300+300+125...)
  current_total_points int 
  
  // Calculated: ceil(total_points / limit)
  required_educators int 
  
  is_compliant boolean // True/False alert
}

// ==========================================
// 3. CHILDREN & PORTFOLIOS (Updated)
// ==========================================

Table children {
  child_id uuid [pk]
  center_id uuid [ref: > centers.center_id]
  assigned_group_id uuid [ref: > groups.group_id]
  
  first_name varchar
  last_name varchar
  birth_date date
  
  // NB Ratio Weight: 300, 200, 125, 100, or 1
  current_ratio_weight int 
  
  // Medical & Diet (JSON allows flexibility)
  medical_info jsonb 
  
  profile_photo_url text
  status varchar // active, inactive
}

Table parent_child {
  id uuid [pk]
  parent_id uuid [ref: > parents.user_id]
  child_id uuid [ref: > children.child_id]
  relationship varchar // Mother, Father, Guardian
  is_emergency_contact boolean
}

// --- PORTFOLIO SYSTEM ---

// The Activity Record
Table portfolio_activities {
  activity_id uuid [pk]
  child_id uuid [ref: > children.child_id]
  
  title varchar // "Sensory Play with Sand"
  description text
  activity_date date
  
  // JSON Array of tags: ["Motor Skills", "Social"]
  learning_domains jsonb 
  
  created_at timestamp
}

// List of Educators who did this activity
Table activity_educators {
  id uuid [pk]
  activity_id uuid [ref: > portfolio_activities.activity_id]
  educator_id uuid [ref: > educators.user_id] // Link to Educator Table
}

// List of Images/Videos for this activity
Table activity_media {
  media_id uuid [pk]
  activity_id uuid [ref: > portfolio_activities.activity_id]
  file_url text
  media_type varchar // 'image', 'video'
  uploaded_at timestamp
}

// ==========================================
// 4. ATTENDANCE & RATIOS
// ==========================================

Table attendance_logs {
  log_id uuid [pk]
  child_id uuid [ref: > children.child_id]
  group_id uuid [ref: > groups.group_id]
  
  check_in_time timestamp
  check_out_time timestamp
  
  // Snapshot of points (e.g. 300) at that specific moment
  ratio_weight_snapshot int 
}

Table staff_assignments {
  assignment_id uuid [pk]
  educator_id uuid [ref: > educators.user_id]
  group_id uuid [ref: > groups.group_id]
  
  check_in_time timestamp
  check_out_time timestamp
  
  is_active boolean
}

// ==========================================
// 5. DYNAMIC SIGNED REPORTS
// ==========================================

Table reports {
  report_id uuid [pk]
  center_id uuid [ref: > centers.center_id]
  
  // Optional link to child (e.g. Incident) or None (e.g. Maintenance)
  child_id uuid [ref: > children.child_id, null]
  
  title varchar 
  type varchar // 'incident', 'daily_log', 'injury'
  
  // The User who started the report
  initiator_id uuid [ref: > users.user_id] 
  
  // The Form Content (JSON)
  data jsonb 
  
  status varchar // 'pending_signatures', 'completed'
  created_at timestamp
}

Table report_signatures {
  sig_id uuid [pk]
  report_id uuid [ref: > reports.report_id]
  
  // The specific person required to sign
  signer_user_id uuid [ref: > users.user_id] 
  
  // Role context: 'parent', 'director', 'witness'
  role_required varchar 
  
  signed_at timestamp // Null = Pending
  signature_token text // Digital proof
}

// ==========================================
// 6. COMMUNICATION & SAFETY
// ==========================================

Table notifications {
  notif_id uuid [pk]
  user_id uuid [ref: > users.user_id]
  title varchar
  message text
  is_read boolean
  read_at timestamp
  created_at timestamp
}

Table evacuation_drills {
  drill_id uuid [pk]
  center_id uuid [ref: > centers.center_id]
  type varchar // 'fire', 'lockdown'
  started_at timestamp
  ended_at timestamp
}

Table evacuation_checks {
  check_id uuid [pk]
  drill_id uuid [ref: > evacuation_drills.drill_id]
  child_id uuid [ref: > children.child_id]
  status varchar // 'safe', 'missing'
  checked_at timestamp
}
